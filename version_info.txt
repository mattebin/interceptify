# Windows version resource. Without it the .exe shows blank Details in
# Properties and no publisher string anywhere, which for an UNSIGNED binary that
# self-updates is the worst combination: nothing on the file says what it is, so
# a user has no way to tell the real one from something wearing its filename.
# It is not a substitute for signing - it is the part we can do today.
#
# filevers/prodvers must match APP_VERSION; tests/test_version_consistency.py
# checks it alongside main.py and interceptify.manifest.
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(2, 0, 5, 0),
    prodvers=(2, 0, 5, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0),
  ),
  kids=[
    StringFileInfo([
      StringTable(
        "040904B0",
        [StringStruct("CompanyName", "mattebin"),
         StringStruct("FileDescription", "Interceptify - Spotify desktop ad blocker"),
         StringStruct("FileVersion", "2.0.5.0"),
         StringStruct("InternalName", "Interceptify"),
         StringStruct("LegalCopyright", "MIT License"),
         StringStruct("OriginalFilename", "Interceptify.exe"),
         StringStruct("ProductName", "Interceptify"),
         StringStruct("ProductVersion", "2.0.5.0")])
    ]),
    VarFileInfo([VarStruct("Translation", [1033, 1200])])
  ]
)
